      1. Beginner: What Are AI Agents?
    </h2>
    <div className="space-y-6 text-[16px] leading-8 text-slate-300">
      <div>

          <code>
            <span className="block text-slate-500"># Chatbot vs agent: the agent loops (decide -&gt; use a tool -&gt; look at the result -&gt; repeat)</span>
            <span className="block">PRICES = &#123;&quot;laptop&quot;: 900, &quot;phone&quot;: 600, &quot;tablet&quot;: 450&#125;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def tool&#95;cheapest():</span>
            <span className="block">    name = min(PRICES, key=PRICES.get)</span>
            <span className="block">    return name, PRICES&#91;name&#93;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def tool&#95;percent(value, pct):</span>
            <span className="block">    return round(value &#42; pct / 100, 2)</span>
            <span className="block">&nbsp;</span>
            <span className="block">TOOLS = &#123;&quot;cheapest&quot;: tool&#95;cheapest, &quot;percent&quot;: tool&#95;percent&#125;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def chatbot(question):</span>
            <span className="block text-slate-500">    # One call, no tools, no loop: it can only answer from what it already knows</span>

            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def fake&#95;llm(goal, history):</span>
            <span className="block text-slate-500">    # Stands in for a real model: reads what happened so far and picks the next step</span>
