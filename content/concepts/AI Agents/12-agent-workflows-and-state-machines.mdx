    <h2 data-type="code" className="mb-6 border-l-4 border-teal-500 pl-4 text-2xl font-bold tracking-tight text-white sm:text-3xl">
      7. Python Implementation: A State Machine With Parallel Steps and Checkpoints
    </h2>
    <div className="space-y-6 text-[16px] leading-8 text-slate-300">
      <div>
        This script implements a four-node workflow. It branches on the question type, searches three sources in parallel, loops between draft and review, saves a checkpoint after every node, simulates a crash, and resumes from the checkpoint.
      </div>
      <div className="overflow-hidden rounded-2xl border border-white/10 bg-[#060911]">
        <div className="flex items-center justify-between border-b border-white/10 bg-white/[0.03] px-5 py-3">
          <span className="font-mono text-xs font-semibold text-teal-400">workflow_state_machine.py</span>
          <span className="font-mono text-[11px] text-slate-400">Python 3.10+ · standard library only</span>
        </div>
        <pre className="overflow-x-auto p-6 font-mono text-sm leading-7 text-slate-200">
          <code>
            <span className="block">import json, os, tempfile</span>
            <span className="block">from concurrent.futures import ThreadPoolExecutor</span>
            <span className="block">&nbsp;</span>
            <span className="block">CHECKPOINT = os.path.join(tempfile.mkdtemp(), &quot;run.json&quot;)</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def save(state, next&#95;node):</span>
            <span className="block">    with open(CHECKPOINT, &quot;w&quot;) as f:</span>
            <span className="block">        json.dump(&#123;&quot;state&quot;: state, &quot;next&quot;: next&#95;node&#125;, f)</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def load():</span>
            <span className="block">    with open(CHECKPOINT) as f:</span>
            <span className="block">        data = json.load(f)</span>
            <span className="block">    return data&#91;&quot;state&quot;&#93;, data&#91;&quot;next&quot;&#93;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-slate-500"># Each node does some work and names the next node (that is the state transition)</span>
            <span className="block text-cyan-300">def classify(s):</span>
            <span className="block">    s&#91;&quot;kind&quot;&#93; = &quot;billing&quot; if &quot;invoice&quot; in s&#91;&quot;question&quot;&#93; else &quot;technical&quot;      # conditional branch</span>
            <span className="block">    return &quot;research&quot;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def research(s):</span>
            <span className="block">    sources = &#91;&quot;knowledge&#95;base&quot;, &quot;faq&quot;, &quot;tickets&quot;&#93;</span>
            <span className="block">    with ThreadPoolExecutor(max&#95;workers=3) as pool:                          # parallel execution</span>
            <span className="block">        s&#91;&quot;facts&quot;&#93; = list(pool.map(lambda name: &quot;fact from &quot; + name, sources))</span>
            <span className="block">    return &quot;draft&quot;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def draft(s):</span>
            <span className="block">    s&#91;&quot;attempts&quot;&#93; = s.get(&quot;attempts&quot;, 0) + 1</span>
            <span className="block">    s&#91;&quot;draft&quot;&#93; = &quot;Answer built from &quot; + str(len(s&#91;&quot;facts&quot;&#93;)) + &quot; facts&quot;</span>
            <span className="block">    return &quot;review&quot;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def review(s):</span>
            <span className="block">    return &quot;done&quot; if s&#91;&quot;attempts&quot;&#93; &gt;= 2 else &quot;draft&quot;                         # first review rejects: loop back</span>
            <span className="block">&nbsp;</span>
            <span className="block">NODES = &#123;&quot;classify&quot;: classify, &quot;research&quot;: research, &quot;draft&quot;: draft, &quot;review&quot;: review&#125;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def run(state, node, crash&#95;after=None):</span>
            <span className="block">    while node != &quot;done&quot;:</span>
            <span className="block">        finished = node</span>
            <span className="block">        node = NODES&#91;node&#93;(state)</span>
            <span className="block">        save(state, node)                                                    # checkpoint after every node</span>
            <span className="block">        print(&quot;ran&quot;, finished.ljust(9), &quot;-&gt; next:&quot;, node)</span>
            <span className="block">        if finished == crash&#95;after:</span>
            <span className="block">            print(&quot;&#42;&#42;&#42; process crashed &#42;&#42;&#42;&quot;)</span>
            <span className="block">            return None</span>
            <span className="block">    return state</span>
            <span className="block">&nbsp;</span>
            <span className="block">state = &#123;&quot;question&quot;: &quot;Why was my invoice charged twice?&quot;&#125;</span>
            <span className="block">run(state, &quot;classify&quot;, crash&#95;after=&quot;research&quot;)</span>
            <span className="block">&nbsp;</span>
            <span className="block">print(&quot;--- restart: load the checkpoint and continue ---&quot;)</span>
