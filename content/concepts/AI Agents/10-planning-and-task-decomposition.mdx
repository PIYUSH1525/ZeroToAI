    <div className="space-y-6 text-[16px] leading-8 text-slate-300">

        <pre className="overflow-x-auto p-6 font-mono text-sm leading-7 text-slate-200">
          <code>
            <span className="block">from graphlib import TopologicalSorter</span>
            <span className="block">&nbsp;</span>
            <span className="block text-slate-500"># A plan is a set of subtasks and what each one depends on</span>
            <span className="block">plan = &#123;</span>
            <span className="block">    &quot;find&#95;market&#95;data&quot;: &#91;&#93;,</span>
            <span className="block">    &quot;find&#95;tech&#95;data&quot;: &#91;&#93;,</span>
            <span className="block">    &quot;compare&quot;: &#91;&quot;find&#95;market&#95;data&quot;, &quot;find&#95;tech&#95;data&quot;&#93;,</span>
            <span className="block">    &quot;write&#95;report&quot;: &#91;&quot;compare&quot;&#93;,</span>
            <span className="block">&#125;</span>
            <span className="block">completed = set()</span>
            <span className="block">broken = &#123;&quot;find&#95;market&#95;data&quot;&#125;            # this subtask will fail the first time</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def execute(plan, completed, broken):</span>
            <span className="block">    ts = TopologicalSorter(plan)</span>
            <span className="block">    ts.prepare()</span>
            <span className="block">    while ts.is&#95;active():</span>
            <span className="block">        failed = &#91;&#93;</span>
            <span className="block">        for task in sorted(ts.get&#95;ready()):</span>
            <span className="block">            if task in completed:</span>
            <span className="block">                ts.done(task)            # already finished earlier: skip</span>
            <span className="block">            elif task in broken:</span>
            <span className="block">                failed.append(task)</span>
            <span className="block">            else:</span>
            <span className="block">                print(&quot;  ran:&quot;, task)</span>
            <span className="block">                completed.add(task)</span>
            <span className="block">                ts.done(task)</span>
            <span className="block">        if failed:</span>
            <span className="block">            return failed</span>
            <span className="block">    return &#91;&#93;</span>
            <span className="block">&nbsp;</span>
            <span className="block">print(&quot;Plan 1 order if nothing fails:&quot;, list(TopologicalSorter(plan).static&#95;order()))</span>
            <span className="block">print(&quot;Executing plan 1&quot;)</span>
            <span className="block">failed = execute(plan, completed, broken)</span>
            <span className="block">print(&quot;  FAILED:&quot;, failed)</span>
            <span className="block">&nbsp;</span>
            <span className="block text-slate-500"># Replanning: swap the failed subtask for a backup, keep the finished work</span>
            <span className="block text-cyan-300">def replace(plan, old, new):</span>
            <span className="block">    return &#123;(new if k == old else k): &#91;new if d == old else d for d in deps&#93; for k, deps in plan.items()&#125;</span>
