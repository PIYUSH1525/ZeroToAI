    <div className="space-y-6 text-[16px] leading-8 text-slate-300">

          <span className="font-mono text-xs font-semibold text-teal-400">agentic_rag.py</span>
          <span className="font-mono text-[11px] text-slate-400">Python 3.10+ · standard library only</span>
        </div>
        <pre className="overflow-x-auto p-6 font-mono text-sm leading-7 text-slate-200">
          <code>
            <span className="block">DOCS = &#123;</span>
            <span className="block">    &quot;refund&quot;: &quot;Refunds are accepted within 30 days of purchase with a receipt.&quot;,</span>
            <span className="block">    &quot;shipping&quot;: &quot;Standard shipping takes 3 to 5 business days. Express takes 1 to 2 days.&quot;,</span>
            <span className="block">    &quot;warranty&quot;: &quot;All devices include a 12 month warranty covering manufacturing defects.&quot;,</span>
            <span className="block">&#125;</span>
            <span className="block">STOP = &#123;&quot;the&quot;, &quot;a&quot;, &quot;an&quot;, &quot;is&quot;, &quot;of&quot;, &quot;to&quot;, &quot;my&quot;, &quot;i&quot;, &quot;can&quot;, &quot;how&quot;, &quot;what&quot;, &quot;until&quot;, &quot;does&quot;, &quot;do&quot;, &quot;for&quot;, &quot;in&quot;&#125;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def words(text):</span>
            <span className="block">    tokens = &#91;w.strip(&quot;.,?!&apos;&quot;).lower() for w in text.split()&#93;</span>
            <span className="block">    tokens = &#91;w&#91;:-1&#93; if len(w) &gt; 3 and w.endswith(&quot;s&quot;) else w for w in tokens&#93;</span>
            <span className="block">    return &#123;w for w in tokens if w not in STOP&#125;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def retrieve(query):</span>
            <span className="block">    q = words(query)</span>
            <span className="block">    scored = &#91;(len(q &amp; words(text)) / max(len(q), 1), doc&#95;id) for doc&#95;id, text in DOCS.items()&#93;</span>
            <span className="block">    return max(scored)</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def plain&#95;rag(question):</span>
            <span className="block">    score, doc&#95;id = retrieve(question)           # always uses the top result, good or bad</span>
            <span className="block">    return doc&#95;id, round(score, 2)</span>
            <span className="block">&nbsp;</span>
            <span className="block">SYNONYMS = &#123;&quot;arrive&quot;: &quot;shipping&quot;, &quot;send back&quot;: &quot;refund&quot;, &quot;guarantee&quot;: &quot;warranty&quot;, &quot;broken&quot;: &quot;warranty&quot;&#125;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def rewrite(question):</span>
            <span className="block">    extra = &#91;target for phrase, target in SYNONYMS.items() if phrase in question.lower()&#93;</span>
            <span className="block">    return question + &quot; &quot; + &quot; &quot;.join(extra)</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def needs&#95;retrieval(question):</span>
            <span className="block">    return not question.lower().startswith((&quot;hello&quot;, &quot;hi&quot;, &quot;thanks&quot;))</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def agentic&#95;rag(question, threshold=0.2, max&#95;tries=2):</span>
            <span className="block">    if not needs&#95;retrieval(question):</span>
            <span className="block">        return &quot;no retrieval needed, answer directly&quot;</span>
