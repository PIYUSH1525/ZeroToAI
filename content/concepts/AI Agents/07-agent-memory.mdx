  <section>
    <h2 data-type="code" className="mb-6 border-l-4 border-teal-500 pl-4 text-2xl font-bold tracking-tight text-white sm:text-3xl">
      7. Python Implementation: Short-Term, Long-Term and Persistent Memory
    </h2>
    <div className="space-y-6 text-[16px] leading-8 text-slate-300">
      <div>
        This script shows a rolling short-term buffer, a long-term memory that recalls facts by similarity, and persistence to a JSON file that survives a restart. Word overlap stands in for embeddings so it runs with no libraries.
      </div>
      <div className="overflow-hidden rounded-2xl border border-white/10 bg-[#060911]">
        <div className="flex items-center justify-between border-b border-white/10 bg-white/[0.03] px-5 py-3">
          <span className="font-mono text-xs font-semibold text-teal-400">agent_memory.py</span>
          <span className="font-mono text-[11px] text-slate-400">Python 3.10+ · standard library only</span>
        </div>
        <pre className="overflow-x-auto p-6 font-mono text-sm leading-7 text-slate-200">
          <code>
            <span className="block">import json, math, os, tempfile</span>
            <span className="block">from collections import Counter, deque</span>
            <span className="block">&nbsp;</span>
            <span className="block text-slate-500"># 1. Short-term memory: only the most recent messages (a rolling window)</span>
            <span className="block">short&#95;term = deque(maxlen=4)</span>
            <span className="block">for message in &#91;&quot;hi&quot;, &quot;my name is Ana&quot;, &quot;I like hiking&quot;, &quot;what is RAG?&quot;, &quot;thanks&quot;&#93;:</span>
            <span className="block">    short&#95;term.append(message)</span>
            <span className="block">print(&quot;Short-term memory keeps the last 4:&quot;, list(short&#95;term))</span>
            <span className="block">&nbsp;</span>
            <span className="block text-slate-500"># 2. Long-term semantic memory: store facts, recall by meaning</span>
            <span className="block text-slate-500"># (Word overlap stands in for embeddings here. Real systems compare embedding vectors.)</span>
            <span className="block">STOP = &#123;&quot;the&quot;, &quot;a&quot;, &quot;an&quot;, &quot;is&quot;, &quot;of&quot;, &quot;in&quot;, &quot;to&quot;, &quot;does&quot;, &quot;what&quot;, &quot;which&quot;, &quot;who&quot;, &quot;where&quot;&#125;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def vectorize(text):</span>
            <span className="block">    words = &#91;w.strip(&quot;.,?!&quot;).lower() for w in text.split()&#93;</span>
            <span className="block">    words = &#91;w&#91;:-1&#93; if len(w) &gt; 3 and w.endswith(&quot;s&quot;) else w for w in words&#93;</span>
            <span className="block">    return Counter(w for w in words if w not in STOP)</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def cosine(a, b):</span>
            <span className="block">    dot = sum(a&#91;w&#93; &#42; b&#91;w&#93; for w in a)</span>
            <span className="block">    norm = math.sqrt(sum(v &#42; v for v in a.values())) &#42; math.sqrt(sum(v &#42; v for v in b.values()))</span>
            <span className="block">    return dot / norm if norm else 0.0</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">class LongTermMemory:</span>
            <span className="block text-cyan-300">    def &#95;&#95;init&#95;&#95;(self):</span>
            <span className="block">        self.items = &#91;&#93;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">    def remember(self, text):</span>
            <span className="block">        self.items.append(text)</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">    def recall(self, query, k=1):</span>
            <span className="block">        ranked = sorted(self.items, key=lambda item: cosine(vectorize(query), vectorize(item)), reverse=True)</span>
            <span className="block">        return ranked&#91;:k&#93;</span>
            <span className="block">&nbsp;</span>
            <span className="block">memory = LongTermMemory()</span>
            <span className="block">for fact in &#91;&quot;The user is allergic to peanuts&quot;, &quot;The user lives in the city of Pune&quot;,</span>
            <span className="block">             &quot;The user prefers short answers&quot;, &quot;The user is building a recipe app&quot;&#93;:</span>
            <span className="block">    memory.remember(fact)</span>
            <span className="block">&nbsp;</span>
            <span className="block">for question in &#91;&quot;What is the user allergic to?&quot;, &quot;Which city does the user live in?&quot;, &quot;What is the user building?&quot;&#93;:</span>
            <span className="block">    print(question.ljust(36), &quot;-&gt;&quot;, memory.recall(question))</span>
            <span className="block">&nbsp;</span>
            <span className="block text-slate-500"># 3. Persistent state: survives a restart because it is saved to disk</span>
            <span className="block">path = os.path.join(tempfile.mkdtemp(), &quot;memory.json&quot;)</span>
            <span className="block">with open(path, &quot;w&quot;) as f:</span>
            <span className="block">    json.dump(memory.items, f)</span>
            <span className="block">&nbsp;</span>
            <span className="block">restored = LongTermMemory()</span>
