          <div className="text-sm leading-relaxed"><strong className="text-white">Reason:</strong> The LLM decides what to do next, or decides that the goal is reached.</div>
        </div>

          <code>

            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def observe():</span>
            <span className="block">    return room&#91;&quot;temp&quot;&#93;</span>
            <span className="block">&nbsp;</span>
            <span className="block text-cyan-300">def reason(observation, target):</span>

            <span className="block">&nbsp;</span>
            <span className="block">while state&#91;&quot;status&quot;&#93; == &quot;running&quot;:</span>
            <span className="block">    obs = observe()                                   # 1. Observe</span>
            <span className="block">    state&#91;&quot;observations&quot;&#93;.append(obs)</span>
            <span className="block">    decision = reason(obs, state&#91;&quot;goal&quot;&#93;)             # 2. Reason</span>
