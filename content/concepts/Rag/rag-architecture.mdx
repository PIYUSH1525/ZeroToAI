            <span className="block text-cyan-300">class RagSystem:</span>
            <span className="block">    def __init__(self):</span>

            <span className="block">    # PHASE 1: OFFLINE INDEXING (Load -&gt; Chunk -&gt; Embed -&gt; Store)</span>
            <span className="block text-violet-300">    def indexDocument(self, sourceName: str, fullText: str):</span>
